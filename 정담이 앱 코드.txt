  function renderRealisticSampleCanvas(canvas, photoId) {
       if (!canvas) return;
       const ctx = canvas.getContext('2d');
       const w = canvas.width;
       const h = canvas.height;
       ctx.clearRect(0, 0, w, h);


       if (photoId === 'sample_sea_sunset') {
        // 하늘 노을 그라데이션
        const sky = ctx.createLinearGradient(0, 0, 0, h * 0.62);
        sky.addColorStop(0, '#3B4D7A');
        sky.addColorStop(0.45, '#C86D51');
        sky.addColorStop(0.8, '#F2A65A');
        sky.addColorStop(1, '#FDE2A7');
        ctx.fillStyle = sky;
        ctx.fillRect(0, 0, w, h * 0.62);


        // 태양 글로우
        const sunX = w * 0.66;
        const sunY = h * 0.54;
        const sunGrad = ctx.createRadialGradient(sunX, sunY, 6, sunX, sunY,
110);
        sunGrad.addColorStop(0, 'rgba(255, 250, 224, 0.98)');
        sunGrad.addColorStop(0.35, 'rgba(255, 204, 112, 0.65)');
        sunGrad.addColorStop(1, 'rgba(255, 160, 90, 0)');
        ctx.fillStyle = sunGrad;
        ctx.beginPath();
        ctx.arc(sunX, sunY, 110, 0, Math.PI * 2);
        ctx.fill();


        // 구름 층
        ctx.fillStyle = 'rgba(255, 214, 170, 0.28)';
        for (let i = 0; i < 6; i++) {
            ctx.beginPath();
            ctx.ellipse(w * (0.15 + i * 0.16), h * (0.24 + (i % 3) * 0.08), 85,
14, 0, 0, Math.PI * 2);
            ctx.fill();
        }


        // 바다 수면 & 노을 윤슬
        const sea = ctx.createLinearGradient(0, h * 0.6, 0, h * 0.84);
        sea.addColorStop(0, '#7A5C61');
        sea.addColorStop(0.5, '#3E5C76');
        sea.addColorStop(1, '#263B4D');
        ctx.fillStyle = sea;
        ctx.fillRect(0, h * 0.6, w, h * 0.24);


        for (let y = h * 0.62; y < h * 0.83; y += 6) {
            const spread = (y - h * 0.6) * 0.55 + 24;
            const glintGrad = ctx.createLinearGradient(sunX - spread, y, sunX +
spread, y);
            glintGrad.addColorStop(0, 'rgba(255,225,150,0)');
            glintGrad.addColorStop(0.5, 'rgba(255,235,180,0.55)');
            glintGrad.addColorStop(1, 'rgba(255,225,150,0)');
            ctx.fillStyle = glintGrad;
            ctx.fillRect(sunX - spread, y, spread * 2, 2.5);
        }


        // 백사장 모래사장
        const sand = ctx.createLinearGradient(0, h * 0.78, 0, h);
        sand.addColorStop(0, '#C2A382');
        sand.addColorStop(0.35, '#DFC39B');
        sand.addColorStop(1, '#B89772');
        ctx.fillStyle = sand;
        ctx.beginPath();
        ctx.moveTo(0, h * 0.82);
        ctx.quadraticCurveTo(w * 0.45, h * 0.76, w, h * 0.86);
        ctx.lineTo(w, h);
        ctx.lineTo(0, h);
        ctx.closePath();
        ctx.fill();


        // 파도 거품 라인
        ctx.strokeStyle = 'rgba(255, 250, 240, 0.65)';
        ctx.lineWidth = 3;
        ctx.beginPath();
        ctx.moveTo(0, h * 0.825);
        ctx.quadraticCurveTo(w * 0.45, h * 0.768, w, h * 0.865);
        ctx.stroke();


      } else if (photoId === 'sample_spring_blossom') {
        // 맑은 봄 하늘 & 나무 캐노피
const sky = ctx.createLinearGradient(0, 0, 0, h * 0.7);
sky.addColorStop(0, '#8EC5E8');
sky.addColorStop(0.65, '#DCEEF8');
sky.addColorStop(1, '#F5EFE2');
ctx.fillStyle = sky;
ctx.fillRect(0, 0, w, h * 0.7);


// 나뭇가지 실루엣
ctx.strokeStyle = '#5C4638';
ctx.lineCap = 'round';
ctx.lineWidth = 14;
ctx.beginPath();
ctx.moveTo(w * 0.14, h * 0.75);
ctx.quadraticCurveTo(w * 0.22, h * 0.42, w * 0.38, h * 0.18);

  // 몸이 안좋음 / 아픔 키워드 판별 (보호자 요약 붉은색/초록색 강조용)
  function detectHealthDistress(text) {
      const cleaned = (text || '').trim();
      if (!cleaned) return { isDistressed: false, reason: null };


      // '안 아파요', '불편한 곳 없어요', '괜찮아요', '아픈 곳 없' 같은 긍정/무증상
표현 먼저 체크
      const noPainPatterns = [
           /안\s*아[파팠]/,
           /아픈\s*곳\s*(없|안)/,
           /불편한\s*곳\s*(없|안)/,
           /이상\s*(없|무)/,
           /괜찮(아|아요|습니다|은\s*편)/,
           /멀쩡/,
           /건강(해|해요|합니다)/,
           /기운\s*(좋|있)/
      ];


      const distressPatterns = [
           /아[파팠]/,
           /아프/,
           /몸살/,
           /어지러/,
           /어지럼/,
           /두통/,
           /머리.*아/,
           /가슴.*(아|답답|통증)/,
           /숨.*(차|가쁘|곤란)/,
           /기운.*(없|빠져|딸려)/,
           /힘이\s*(없|안\s*나|빠져)/,
           /불편(해|해요|합니다|하네)/,
           /통증/,
           /구토/,
             /메스꺼/,
             /열이?\s*나/,
             /잠.*못\s*자/,
             /소화.*안/,
             /배가?\s*(아|살살)/,
             /허리.*아/,
             /무릎.*아/,
             /다리.*(아|저려|부어)/
        ];


        const hasDistressWord = distressPatterns.some((re) => re.test(cleaned));
        const hasExplicitNoPain = noPainPatterns.some((re) => re.test(cleaned));


        if (hasDistressWord && !hasExplicitNoPain) {
             return {
                  isDistressed: true,
                  reason: cleaned.slice(0, 120)
             };
        }
        return { isDistressed: false, reason: null };
    }


    function parseMedicationAnswer(text) {
        const t = (text || '').trim();
        if (/안\s*(먹|드|복용)/.test(t) || /못\s*(먹|드|챙)/.test(t) || /깜빡/.test(t) || /잊
어/.test(t)) {
             return 'not_taken';
        }
        if (/(먹었|드셨|복용|챙겼|네|예|완료)/.test(t)) {
             return 'taken';
        }
        return 'not_recorded';
    }


    function parseConditionAnswer(text, isDistressed) {
        if (isDistressed) return 'unwell';
        const t = (text || '').trim();
         if (/(좋|상쾌|개운|괜찮|건강|활기)/.test(t)) return 'good';
         if (/(보통|그냥|무난)/.test(t)) return 'okay';
         return 'not_recorded';
     }

    async function callGeminiText(promptText) {
         const key = (state.settings.geminiApiKey || '').trim();
         if (state.settings.forceFallback || !key) {
             throw new Error('FALLBACK_MODE');
        }


        const controller = new AbortController();
        const timer = setTimeout(() => controller.abort(), 12000);


        try {
            const res = await fetch(


`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:gener
ateContent?key=${encodeURIComponent(key)}`,
                 {
                     method: 'POST',
                     headers: { 'Content-Type': 'application/json' },
                     signal: controller.signal,
                     body: JSON.stringify({
                          contents: [{ parts: [{ text: promptText }] }],
                          generationConfig: { temperature: 0.5, maxOutputTokens: 400 }
                     })
                 }
            );
            clearTimeout(timer);
            if (!res.ok) {
                 throw new Error(`HTTP_${res.status}`);
            }
            const data = await res.json();
            const text = data?.candidates?.[0]?.content?.parts?.[0]?.text || '';
            if (!text.trim()) throw new Error('EMPTY_AI_RESPONSE');
            return text.trim();
        } catch (err) {
            clearTimeout(timer);
            throw err;
        }
    }

   function openSettingsModal() {
        const modal = document.getElementById('settings-modal');
        const keyInput = document.getElementById('input-gemini-key');
        const fallbackToggle = document.getElementById('toggle-force-fallback');
        const countEl = document.getElementById('modal-session-count');
        if (keyInput) keyInput.value = state.settings.geminiApiKey || '';
        if                (fallbackToggle)        fallbackToggle.checked           =
Boolean(state.settings.forceFallback);
        if (countEl) countEl.textContent = `${state.sessions.length}개`;
        if (modal) modal.classList.remove('hidden');
    }


    function closeSettingsModal() {
        const modal = document.getElementById('settings-modal');
        if (modal) modal.classList.add('hidden');
    }


    function saveSettingsFromModal() {
        const keyInput = document.getElementById('input-gemini-key');
        const fallbackToggle = document.getElementById('toggle-force-fallback');
        state.settings.geminiApiKey = keyInput ? keyInput.value.trim() : '';
        state.settings.forceFallback          =            fallbackToggle          ?
 Boolean(fallbackToggle.checked) : false;
        saveState();
        closeSettingsModal();
        showToast('AI 연결 및 시연 설정이 저장되었습니다.', 'success');
        render();
    }

     function showToast(message, type = 'info') {
         const container = document.getElementById('toast-container');
         if (!container) return;


         const colorMap = {
               success: 'bg-[#1E5A40] text-white border-[#2E7D5B]',
               error: 'bg-red-700 text-white border-red-900',
               warn: 'bg-[#B85226] text-white border-[#E06D3B]',
               info: 'bg-gray-900 text-white border-gray-700'
         };


         const toast = document.createElement('div');
         toast.className = `pointer-events-auto px-5 py-3.5 rounded-2xl border-2
shadow-xl text-base font-bold transition transform ${
               colorMap[type] || colorMap.info
    }`;
    toast.textContent = message;
    container.appendChild(toast);


    setTimeout(() => {
          toast.style.opacity = '0';
          toast.style.transform = 'translateY(8px)';
          setTimeout(() => toast.remove(), 220);
    }, 3400);
}


function formatKoreanDateTime(d) {
    const month = d.getMonth() + 1;
    const day = d.getDate();
    const hour = String(d.getHours()).padStart(2, '0');
    const min = String(d.getMinutes()).padStart(2, '0');
    return `${month}월 ${day}일 ${hour}:${min}`;
}


function escapeHtml(str) {
    if (str === null || str === undefined) return '';
    return String(str)
          .replace(/&/g, '&amp;')
          .replace(/</g, '&lt;')
          .replace(/>/g, '&gt;')
          .replace(/"/g, '&quot;')
          .replace(/'/g, '&#39;');
}


function scrollChatToBottom() {
    setTimeout(() => {
          const list = document.getElementById('chat-message-list');
          if (list) list.scrollTop = list.scrollHeight;
    }, 50);
}
